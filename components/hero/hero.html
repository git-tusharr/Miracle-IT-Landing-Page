-- Miracle IT Data Analytics & SQL Engineering
WITH student_growth AS (
  SELECT 
    student_id,
    track_name,
    github_commits,
    AVG(lab_score) OVER (PARTITION BY track_name) AS avg_lab_rating
  FROM miracle_academy_cohort_2026
  WHERE center_location = 'M.P. Nagar, Bhopal'
)
SELECT track_name, COUNT(*) AS job_ready_students
FROM student_growth
GROUP BY track_name;
